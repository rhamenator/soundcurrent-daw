cmake_minimum_required(VERSION 3.20)
project(SoundCurrentDAW VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)

add_library(sc-session src/session.cpp)
add_library(SoundCurrent::Session ALIAS sc-session)
target_include_directories(sc-session PUBLIC include)
target_compile_features(sc-session PUBLIC cxx_std_20)
if(WIN32)
  target_link_libraries(sc-session PRIVATE bcrypt)
endif()
add_library(sc-engine src/eq_coefficients.cpp src/eq.cpp)
add_library(SoundCurrent::Engine ALIAS sc-engine)
target_link_libraries(sc-engine PUBLIC sc-session)
add_library(sc-capture src/capture.cpp)
add_library(SoundCurrent::Capture ALIAS sc-capture)
target_link_libraries(sc-capture PUBLIC sc-session)
add_library(sc-audio-bridge src/audio_bridge.cpp)
add_library(SoundCurrent::AudioBridge ALIAS sc-audio-bridge)
target_link_libraries(sc-audio-bridge PUBLIC sc-engine sc-capture)
add_library(sc-playback src/playback.cpp)
add_library(SoundCurrent::Playback ALIAS sc-playback)
target_link_libraries(sc-playback PUBLIC sc-capture sc-engine)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  option(SC_BUILD_PIPEWIRE "Build the native PipeWire filter adapter" ON)
  if(SC_BUILD_PIPEWIRE)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3>=1.6.2)
    add_library(sc-pipewire src/pipewire_filter.cpp)
    add_library(SoundCurrent::PipeWire ALIAS sc-pipewire)
    target_link_libraries(sc-pipewire PUBLIC sc-audio-bridge PRIVATE PkgConfig::PIPEWIRE)
  endif()
endif()
add_library(sc-project-store src/project_store.cpp)
add_library(SoundCurrent::ProjectStore ALIAS sc-project-store)
target_link_libraries(sc-project-store PUBLIC sc-session)
target_include_directories(sc-project-store SYSTEM PRIVATE third_party)
if(WIN32)
  target_link_libraries(sc-project-store PRIVATE bcrypt)
else()
  find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
  target_link_libraries(sc-project-store PRIVATE OpenSSL::Crypto)
endif()
option(SC_BUILD_MEDIA "Build disk-side RF64 recording (requires libsndfile runtime/library)" ON)
if(SC_BUILD_MEDIA)
  find_library(SNDFILE_LIBRARY NAMES sndfile libsndfile.so.1 REQUIRED)
  find_package(Threads REQUIRED)
  add_library(sc-media src/media_io.cpp src/recording.cpp src/recording_worker.cpp src/playback_reader.cpp src/playback_run.cpp src/playback_bridge.cpp)
  add_library(SoundCurrent::Media ALIAS sc-media)
  target_include_directories(sc-media SYSTEM PRIVATE third_party third_party/libsndfile)
  target_link_libraries(sc-media PUBLIC sc-capture sc-project-store sc-playback sc-audio-bridge PRIVATE ${SNDFILE_LIBRARY} Threads::Threads)
  if(WIN32)
    target_link_libraries(sc-media PRIVATE bcrypt)
  else()
    target_link_libraries(sc-media PRIVATE OpenSSL::Crypto)
  endif()
  if(TARGET sc-pipewire)
    add_library(sc-pipewire-playback src/pipewire_playback.cpp)
    add_library(SoundCurrent::PipeWirePlayback ALIAS sc-pipewire-playback)
    target_link_libraries(sc-pipewire-playback PUBLIC sc-pipewire sc-media)
  endif()
  add_executable(sc-play-tool tools/play_tool.cpp)
  target_link_libraries(sc-play-tool PRIVATE sc-media)
  if(MINGW)
    target_link_options(sc-play-tool PRIVATE -municode)
  endif()
  add_executable(sc-record-tool tools/record_tool.cpp)
  target_link_libraries(sc-record-tool PRIVATE sc-media)
  if(MINGW)
    target_link_options(sc-record-tool PRIVATE -municode)
  endif()
endif()
add_executable(sc-project-tool tools/project_tool.cpp)
target_link_libraries(sc-project-tool PRIVATE sc-project-store)
if(MINGW)
  target_link_options(sc-project-tool PRIVATE -municode)
endif()
option(SC_BUILD_DESKTOP "Build the Qt desktop project editor" OFF)
if(SC_BUILD_DESKTOP)
  if(NOT TARGET sc-media)
    message(FATAL_ERROR "Desktop playback requires SC_BUILD_MEDIA=ON")
  endif()
  find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets)
  add_library(sc-ui-controller ui/project_controller.cpp)
  target_include_directories(sc-ui-controller PUBLIC ui)
  target_link_libraries(sc-ui-controller PUBLIC sc-project-store PRIVATE Qt6::Core)
  add_library(sc-ui-playback ui/playback_controller.cpp)
  target_include_directories(sc-ui-playback PUBLIC ui)
  target_link_libraries(sc-ui-playback PUBLIC sc-media PRIVATE Qt6::Core)
  if(TARGET sc-pipewire-playback)
    target_compile_definitions(sc-ui-playback PRIVATE SC_UI_PIPEWIRE)
    target_link_libraries(sc-ui-playback PRIVATE sc-pipewire-playback)
  endif()
  add_executable(soundcurrent-daw ui/main.cpp ui/studio_window.cpp)
  target_link_libraries(soundcurrent-daw PRIVATE sc-ui-controller sc-ui-playback Qt6::Widgets)
  if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(sc-controller-tests tests/controller_tests.cpp)
    target_link_libraries(sc-controller-tests PRIVATE sc-ui-controller Qt6::Core)
    add_test(NAME desktop-controller COMMAND sc-controller-tests)
    set_tests_properties(desktop-controller PROPERTIES TIMEOUT 60)
    add_executable(sc-playback-controller-tests tests/playback_controller_tests.cpp)
    target_link_libraries(sc-playback-controller-tests PRIVATE sc-ui-playback Qt6::Core)
    add_test(NAME desktop-playback-controller COMMAND sc-playback-controller-tests)
    set_tests_properties(desktop-playback-controller PROPERTIES TIMEOUT 60)
    if(TARGET sc-pipewire-playback)
      add_executable(sc-pipewire-ui-fixture tests/pipewire_ui_fixture.cpp tests/rt_audit.cpp ui/studio_window.cpp)
      target_link_libraries(sc-pipewire-ui-fixture PRIVATE sc-ui-controller sc-ui-playback sc-pipewire-playback Qt6::Widgets Qt6::Test)
      target_include_directories(sc-pipewire-ui-fixture SYSTEM PRIVATE third_party/libsndfile)
    endif()
    add_executable(sc-ui-tests tests/ui_tests.cpp ui/studio_window.cpp)
    target_link_libraries(sc-ui-tests PRIVATE sc-ui-controller sc-ui-playback Qt6::Widgets Qt6::Test)
    add_test(NAME desktop-ui COMMAND sc-ui-tests)
    set_tests_properties(desktop-ui PROPERTIES TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
  install(TARGETS soundcurrent-daw RUNTIME DESTINATION bin)
  install(FILES packaging/soundcurrent-daw.desktop DESTINATION share/applications)
  install(FILES packaging/soundcurrent-daw.svg DESTINATION share/icons/hicolor/scalable/apps)
endif()
if(BUILD_TESTING)
  add_executable(sc-session-tests tests/session_tests.cpp)
  target_link_libraries(sc-session-tests PRIVATE sc-project-store)
  target_include_directories(sc-session-tests SYSTEM PRIVATE third_party)
  add_test(NAME session-state COMMAND sc-session-tests)
  find_package(Threads REQUIRED)
  add_executable(sc-engine-tests tests/engine_tests.cpp tests/rt_audit.cpp)
  target_link_libraries(sc-engine-tests PRIVATE sc-engine Threads::Threads)
  add_executable(sc-immediate-tests tests/immediate_tests.cpp tests/rt_audit.cpp)
  target_link_libraries(sc-immediate-tests PRIVATE sc-engine Threads::Threads)
  add_test(NAME immediate-controls COMMAND sc-immediate-tests)
  set_tests_properties(immediate-controls PROPERTIES TIMEOUT 60)
  add_executable(sc-capture-tests tests/capture_tests.cpp tests/rt_audit.cpp)
  target_link_libraries(sc-capture-tests PRIVATE sc-capture Threads::Threads)
  add_test(NAME capture-transport COMMAND sc-capture-tests)
  add_executable(sc-audio-bridge-tests tests/audio_bridge_tests.cpp tests/rt_audit.cpp)
  target_link_libraries(sc-audio-bridge-tests PRIVATE sc-audio-bridge)
  add_test(NAME audio-bridge COMMAND sc-audio-bridge-tests)
  if(SC_BUILD_MEDIA)
    add_executable(sc-recording-tests tests/recording_tests.cpp tests/rt_audit.cpp)
    target_link_libraries(sc-recording-tests PRIVATE sc-media Threads::Threads)
    target_include_directories(sc-recording-tests SYSTEM PRIVATE third_party third_party/libsndfile)
    add_test(NAME recording-recovery COMMAND sc-recording-tests)
    set_tests_properties(recording-recovery PROPERTIES TIMEOUT 60)
    add_executable(sc-playback-bridge-tests tests/playback_bridge_tests.cpp tests/rt_audit.cpp)
    target_link_libraries(sc-playback-bridge-tests PRIVATE sc-media Threads::Threads)
    add_test(NAME playback-native-clock COMMAND sc-playback-bridge-tests)
    set_tests_properties(playback-native-clock PROPERTIES TIMEOUT 60)
    add_executable(sc-playback-tests tests/playback_tests.cpp tests/rt_audit.cpp)
    target_link_libraries(sc-playback-tests PRIVATE sc-media Threads::Threads)
    target_include_directories(sc-playback-tests SYSTEM PRIVATE third_party/libsndfile)
    add_test(NAME playback-read-ahead COMMAND sc-playback-tests)
    set_tests_properties(playback-read-ahead PROPERTIES TIMEOUT 60)
    if(TARGET sc-pipewire)
      add_executable(sc-pipewire-fixture tests/pipewire_fixture.cpp tests/rt_audit.cpp)
      target_link_libraries(sc-pipewire-fixture PRIVATE sc-pipewire sc-media)
      target_include_directories(sc-pipewire-fixture SYSTEM PRIVATE third_party/libsndfile)
      add_executable(sc-pipewire-playback-fixture tests/pipewire_playback_fixture.cpp tests/rt_audit.cpp)
      target_link_libraries(sc-pipewire-playback-fixture PRIVATE sc-pipewire-playback)
      target_include_directories(sc-pipewire-playback-fixture SYSTEM PRIVATE third_party/libsndfile)
      # Explicitly invoked integration fixture, never an implicit hardware test.
    endif()
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    foreach(test sc-pipewire-ui-fixture sc-playback-bridge-tests sc-immediate-tests sc-pipewire-playback-fixture sc-playback-tests sc-engine-tests sc-capture-tests sc-recording-tests sc-audio-bridge-tests sc-pipewire-fixture)
      if(TARGET ${test})
        target_compile_definitions(${test} PRIVATE SC_WRAP_LIBC)
        target_link_options(${test} PRIVATE
          -Wl,--wrap=malloc -Wl,--wrap=free -Wl,--wrap=calloc -Wl,--wrap=realloc
          -Wl,--wrap=pthread_mutex_lock)
      endif()
    endforeach()
  endif()
  add_test(NAME prepared-eq COMMAND sc-engine-tests)
  set_tests_properties(prepared-eq PROPERTIES TIMEOUT 60)
endif()
foreach(target sc-pipewire-ui-fixture sc-playback-controller-tests sc-ui-playback sc-pipewire-playback sc-playback-bridge-tests sc-ui-controller soundcurrent-daw sc-controller-tests sc-ui-tests sc-immediate-tests sc-play-tool sc-pipewire-playback-fixture sc-playback-tests sc-session sc-engine sc-capture sc-playback sc-audio-bridge sc-pipewire sc-media sc-project-store sc-project-tool sc-record-tool sc-session-tests sc-engine-tests sc-capture-tests sc-recording-tests sc-audio-bridge-tests sc-pipewire-fixture)
  if(TARGET ${target})
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
    endif()
  endif()
endforeach()
