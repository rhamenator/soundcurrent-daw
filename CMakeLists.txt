cmake_minimum_required(VERSION 3.20)
project(SoundCurrentDAW VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)

add_library(sc-session src/session.cpp)
add_library(SoundCurrent::Session ALIAS sc-session)
target_include_directories(sc-session PUBLIC include)
target_compile_features(sc-session PUBLIC cxx_std_20)
if(WIN32)
  target_link_libraries(sc-session PRIVATE bcrypt)
endif()
add_library(sc-engine src/eq_coefficients.cpp src/eq.cpp)
add_library(SoundCurrent::Engine ALIAS sc-engine)
target_link_libraries(sc-engine PUBLIC sc-session)
add_library(sc-capture src/capture.cpp)
add_library(SoundCurrent::Capture ALIAS sc-capture)
target_link_libraries(sc-capture PUBLIC sc-session)
add_library(sc-project-store src/project_store.cpp)
add_library(SoundCurrent::ProjectStore ALIAS sc-project-store)
target_link_libraries(sc-project-store PUBLIC sc-session)
target_include_directories(sc-project-store SYSTEM PRIVATE third_party)
if(WIN32)
  target_link_libraries(sc-project-store PRIVATE bcrypt)
else()
  find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
  target_link_libraries(sc-project-store PRIVATE OpenSSL::Crypto)
endif()
option(SC_BUILD_MEDIA "Build disk-side RF64 recording (requires libsndfile runtime/library)" ON)
if(SC_BUILD_MEDIA)
  find_library(SNDFILE_LIBRARY NAMES sndfile libsndfile.so.1 REQUIRED)
  find_package(Threads REQUIRED)
  add_library(sc-media src/media_io.cpp src/recording.cpp src/recording_worker.cpp)
  add_library(SoundCurrent::Media ALIAS sc-media)
  target_include_directories(sc-media SYSTEM PRIVATE third_party third_party/libsndfile)
  target_link_libraries(sc-media PUBLIC sc-capture sc-project-store PRIVATE ${SNDFILE_LIBRARY} Threads::Threads)
  if(WIN32)
    target_link_libraries(sc-media PRIVATE bcrypt)
  else()
    target_link_libraries(sc-media PRIVATE OpenSSL::Crypto)
  endif()
  add_executable(sc-record-tool tools/record_tool.cpp)
  target_link_libraries(sc-record-tool PRIVATE sc-media)
  if(MINGW)
    target_link_options(sc-record-tool PRIVATE -municode)
  endif()
endif()
add_executable(sc-project-tool tools/project_tool.cpp)
target_link_libraries(sc-project-tool PRIVATE sc-project-store)
if(MINGW)
  target_link_options(sc-project-tool PRIVATE -municode)
endif()
if(BUILD_TESTING)
  add_executable(sc-session-tests tests/session_tests.cpp)
  target_link_libraries(sc-session-tests PRIVATE sc-project-store)
  target_include_directories(sc-session-tests SYSTEM PRIVATE third_party)
  add_test(NAME session-state COMMAND sc-session-tests)
  find_package(Threads REQUIRED)
  add_executable(sc-engine-tests tests/engine_tests.cpp tests/rt_audit.cpp)
  target_link_libraries(sc-engine-tests PRIVATE sc-engine Threads::Threads)
  add_executable(sc-capture-tests tests/capture_tests.cpp tests/rt_audit.cpp)
  target_link_libraries(sc-capture-tests PRIVATE sc-capture Threads::Threads)
  add_test(NAME capture-transport COMMAND sc-capture-tests)
  if(SC_BUILD_MEDIA)
    add_executable(sc-recording-tests tests/recording_tests.cpp tests/rt_audit.cpp)
    target_link_libraries(sc-recording-tests PRIVATE sc-media Threads::Threads)
    target_include_directories(sc-recording-tests SYSTEM PRIVATE third_party third_party/libsndfile)
    add_test(NAME recording-recovery COMMAND sc-recording-tests)
    set_tests_properties(recording-recovery PROPERTIES TIMEOUT 60)
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    foreach(test sc-engine-tests sc-capture-tests sc-recording-tests)
      if(TARGET ${test})
        target_compile_definitions(${test} PRIVATE SC_WRAP_LIBC)
        target_link_options(${test} PRIVATE
          -Wl,--wrap=malloc -Wl,--wrap=free -Wl,--wrap=calloc -Wl,--wrap=realloc
          -Wl,--wrap=pthread_mutex_lock)
      endif()
    endforeach()
  endif()
  add_test(NAME prepared-eq COMMAND sc-engine-tests)
  set_tests_properties(prepared-eq PROPERTIES TIMEOUT 60)
endif()
foreach(target sc-session sc-engine sc-capture sc-media sc-project-store sc-project-tool sc-record-tool sc-session-tests sc-engine-tests sc-capture-tests sc-recording-tests)
  if(TARGET ${target})
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
    endif()
  endif()
endforeach()
