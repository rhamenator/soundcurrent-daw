  # Opt-in diagnostics link unchanged static processing libraries with GNU ABI
  # wrappers. No profiling hook or clock query is added to production libraries.
  function(sc_wrap_processing_stages target)
    target_sources(${target} PRIVATE tests/native_processing_stages.cpp)
    target_compile_definitions(${target} PRIVATE SC_NATIVE_PROCESSING_STAGES)
    if("manual" IN_LIST ARGN)
      target_compile_definitions(${target} PRIVATE SC_NATIVE_MANUAL_STAGES)
      target_link_options(${target} PRIVATE "-Wl,--wrap=_ZN12soundcurrent3daw17ManualPunchBridge7processERKNS0_16DeviceBlockClockESt4spanIKPKfLm18446744073709551615EES5_IKPfLm18446744073709551615EEj")
    endif()
    foreach(symbol
      _ZN12soundcurrent3daw11CapturePipe4pushESt4spanIKPKfLm18446744073709551615EEjl
      _ZN12soundcurrent3daw12EqLiveDriver7processESt4spanIKPKfLm18446744073709551615EES2_IKPfLm18446744073709551615EEj
      _ZN12soundcurrent3daw16PreparedMixGraph7processESt4spanIKS2_IKPKfLm18446744073709551615EELm18446744073709551615EES2_IKPfLm18446744073709551615EEj
      _ZN12soundcurrent3daw12DuplexBridge7processERKNS0_16DeviceBlockClockESt4spanIKPKfLm18446744073709551615EES5_IKPfLm18446744073709551615EEj)
      target_link_options(${target} PRIVATE "-Wl,--wrap=${symbol}")
    endforeach()
  endfunction()
