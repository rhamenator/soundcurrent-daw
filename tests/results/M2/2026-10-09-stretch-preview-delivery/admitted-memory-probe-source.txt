from pathlib import Path
import json,subprocess,struct,math,uuid,tempfile,hashlib
worker=Path('.cache/build-desktop-release/sc-stretch-render-worker').resolve()
with tempfile.TemporaryDirectory(prefix='sc-stretch-memory-admitted-') as temp:
 root=Path(temp);(root/'media').mkdir();jobs=root/'media/derived';jobs.mkdir()
 frames=16384;channels=32;rate=192000
 payload=b''.join(struct.pack('<f',.3*math.sin(2*math.pi*1000*f/rate)) for f in range(frames) for _ in range(channels))
 fmt=struct.pack('<HHIIHH',3,channels,rate,rate*channels*4,channels*4,32)
 data=b'RIFF'+struct.pack('<I',36+len(payload))+b'WAVEfmt '+struct.pack('<I',16)+fmt+b'data'+struct.pack('<I',len(payload))+payload
 (root/'media/raw.wav').write_bytes(data)
 outcomes=[]
 for memory in [32,256]:
  request={'protocol':'sc-stretch-render-v2','operation':str(uuid.uuid4()),'assetId':str(uuid.uuid4()),'relative':'media/raw.wav','sha256':hashlib.sha256(data).hexdigest(),'rate':rate,'channels':channels,'sourceFrames':frames,'first':0,'firstFraction':0,'firstDenominator':1,'frames':frames,'timeNumerator':3,'timeDenominator':2,'pitchMilliCents':1200000,'formantPreserved':False}
  p=subprocess.Popen([str(worker),'--project-root',str(root),'--jobs-root',str(jobs),'--memory-mib',str(memory),'--maximum-input-frames','1000000','--maximum-output-bytes','67108864','--deadline-ms','10000'],stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True)
  p.stdin.write(json.dumps(request));p.stdin.close();line=p.stdout.readline();ready=json.loads(line) if line else {}
  job=jobs/request['operation']
  if ready.get('event')=='ready':(job/'start.request').write_text('start\n')
  output=p.stdout.read();error=p.stderr.read();p.wait(timeout=15)
  outcomes.append({'memoryMiB':memory,'pid':p.pid,'ready':ready,'exitCode':p.returncode,'intent':(job/'intent.json').is_file(),'complete':(job/'complete.json').is_file(),'stdout':output,'stderr':error})
 assert outcomes[0]['exitCode']!=0 and not outcomes[0]['complete'] and not outcomes[0]['intent'] and json.loads(outcomes[0]['stderr'])['messageId']=='stretch.resource_limit',outcomes
 assert outcomes[1]['exitCode']==0 and outcomes[1]['complete'],outcomes
 Path('.cache/stretch-memory-admitted-probe.json').write_text(json.dumps(outcomes,indent=2)+'\n')
 print(json.dumps(outcomes,indent=2))
