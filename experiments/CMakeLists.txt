cmake_minimum_required(VERSION 3.20)
project(soundcurrent_daw_feasibility LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT SOUNDCURRENT_SDK_SOURCE)
  message(FATAL_ERROR "Set SOUNDCURRENT_SDK_SOURCE to the read-only Studio checkout")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${SOUNDCURRENT_SDK_SOURCE}" rev-parse HEAD
  OUTPUT_VARIABLE sdk_revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT sdk_revision STREQUAL "65151a8fec1aa4b4e6c283e4514d7f3d9166fe2b")
  message(FATAL_ERROR "Feasibility source must match the frozen Studio SDK revision")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${SOUNDCURRENT_SDK_SOURCE}" status --porcelain=v1
  OUTPUT_VARIABLE sdk_changes OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(sdk_changes)
  message(FATAL_ERROR "Feasibility source must be clean; do not reset an active Studio checkout")
endif()
set(SOUNDCURRENT_BUILD_DESKTOP OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOUNDCURRENT_SDK_SOURCE}" sdk)
add_executable(eq-feasibility eq_feasibility.cpp)
target_link_libraries(eq-feasibility PRIVATE SoundCurrent::Engine SoundCurrent::Wave)
target_compile_options(eq-feasibility PRIVATE -Wall -Wextra -Wpedantic)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3)
add_executable(pipewire-api-feasibility pipewire_api.cpp)
target_link_libraries(pipewire-api-feasibility PRIVATE PkgConfig::PIPEWIRE)
