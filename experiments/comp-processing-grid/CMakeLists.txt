cmake_minimum_required(VERSION 3.20)
project(sc_comp_processing_grid_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT SC_DAW_BUILD)
  message(FATAL_ERROR "Set SC_DAW_BUILD to the current dedicated DAW build")
endif()
get_filename_component(SC_DAW_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
find_package(Threads REQUIRED)
foreach(component IN ITEMS positioned-resampling session rubberband)
  find_library(probe_${component} NAMES sc-${component}
    PATHS "${SC_DAW_BUILD}" "${SC_DAW_BUILD}/Release" NO_DEFAULT_PATH REQUIRED)
endforeach()
add_executable(sc-comp-grid-probe probe.cpp)
target_include_directories(sc-comp-grid-probe PRIVATE "${SC_DAW_SOURCE}/include")
target_include_directories(sc-comp-grid-probe SYSTEM PRIVATE
  "${SC_DAW_SOURCE}/third_party" "${SC_DAW_SOURCE}/third_party/rubberband")
target_link_libraries(sc-comp-grid-probe PRIVATE
  "${probe_positioned-resampling}" "${probe_session}" "${probe_rubberband}" Threads::Threads)
if(WIN32)
  target_link_libraries(sc-comp-grid-probe PRIVATE bcrypt)
  target_compile_definitions(sc-comp-grid-probe PRIVATE NOMINMAX)
endif()
if(MSVC)
  target_compile_options(sc-comp-grid-probe PRIVATE /W4 /fp:precise)
else()
  target_compile_options(sc-comp-grid-probe PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
endif()
