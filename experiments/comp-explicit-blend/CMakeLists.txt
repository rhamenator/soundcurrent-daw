cmake_minimum_required(VERSION 3.20)
project(sc_comp_blend_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT SC_DAW_BUILD)
  message(FATAL_ERROR "Supply a dedicated current core build")
endif()
get_filename_component(SC_DAW_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
find_package(Threads REQUIRED)
foreach(component IN ITEMS comp-handle-pcm session)
  find_library(probe_${component} NAMES sc-${component} PATHS "${SC_DAW_BUILD}" NO_DEFAULT_PATH REQUIRED)
endforeach()
add_executable(sc-comp-blend-probe probe.cpp)
target_include_directories(sc-comp-blend-probe PRIVATE "${SC_DAW_SOURCE}/include")
target_include_directories(sc-comp-blend-probe SYSTEM PRIVATE "${SC_DAW_SOURCE}/third_party")
target_link_libraries(sc-comp-blend-probe PRIVATE "${probe_comp-handle-pcm}" "${probe_session}" Threads::Threads)
if(WIN32)
  target_compile_definitions(sc-comp-blend-probe PRIVATE NOMINMAX)
  target_link_libraries(sc-comp-blend-probe PRIVATE bcrypt)
endif()
if(MSVC)
  target_compile_options(sc-comp-blend-probe PRIVATE /W4 /fp:precise)
else()
  target_compile_options(sc-comp-blend-probe PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
endif()
