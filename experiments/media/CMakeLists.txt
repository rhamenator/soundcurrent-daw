cmake_minimum_required(VERSION 3.20)
project(SoundCurrentMediaFeasibility LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This descriptor/flush feasibility probe is Linux-specific; Windows media qualification is separate")
endif()
find_library(SNDFILE_LIBRARY NAMES sndfile libsndfile.so.1 REQUIRED)
add_executable(sndfile-feasibility sndfile_feasibility.cpp)
target_include_directories(sndfile-feasibility SYSTEM PRIVATE ../../third_party/libsndfile)
target_link_libraries(sndfile-feasibility PRIVATE ${SNDFILE_LIBRARY})
if(NOT MSVC)
  target_compile_options(sndfile-feasibility PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
